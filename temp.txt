function zamaniGuncelle() {
    let langCode;
    
    // Anlık olarak seçili dili kontrol et
    if (lang.value === "Türkçe") {
        langCode = 'tr-TR';
    } else if (lang.value === "English") {

      month: 'long',
      day: '2-digit',
      weekday: 'long',
      hour: '2-digit',
      minute: '2-digit',
      timeZoneName: 'short'
    });

    // 2. Eksik olan kısım: DOM'u yeni formatlı zaman ile değiştiriyoruz
    dateAndHour.textContent = formattedTime;
